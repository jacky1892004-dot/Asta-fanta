    <script>

        // --- GESTIONE MENU SIDEBAR (20% WIDE) ---

        function toggleMenu() {

            document.getElementById('sidebar').classList.toggle('active');

            document.getElementById('overlay').classList.toggle('active');

        }


        function switchTab(tabId) {

            document.querySelectorAll('.section').forEach(el => el.classList.remove('active'));

            document.querySelectorAll('.sidebar-btn').forEach(el => el.classList.remove('active-tab'));

            

            document.getElementById(tabId).classList.add('active');

            event.target.classList.add('active-tab');

            toggleMenu(); // Chiude il menu dopo il click

        }


        // --- ACCESSO E CREAZIONE STANZA ---

        function generateRoomCode() {

            return Math.random().toString(36).substring(2, 7).toUpperCase();

        }


        function createRoom() {

            const code = generateRoomCode();

            document.getElementById('room-input').value = code;

            alert("Codice stanza generato: " + code + "\nCondividilo con i tuoi amici in Italia!");

        }


        function joinRoom() {

            const user = document.getElementById('username-input').value;

            const room = document.getElementById('room-input').value;

            if(!user || !room) {

                alert("Inserisci sia il nome utente che il codice stanza!");

                return;

            }

            document.getElementById('auth-section').style.display = 'none';

            document.getElementById('app-section').style.display = 'block';

            loadListoneData();

        }


        // --- LOGICA ASTA E PUNTATE ---

        let currentBid = 1;

        function incrementBid() {

            let input = document.getElementById('bid-input');

            input.value = parseInt(input.value) + 1;

        }


        function placeBid() {

            let val = parseInt(document.getElementById('bid-input').value);

            if(val > currentBid) {

                currentBid = val;

                document.getElementById('current-bid').innerText = currentBid + " FM";

                document.getElementById('highest-bidder').innerText = document.getElementById('username-input').value;

                addChatMessage("Sistema", document.getElementById('username-input').value + " ha offerto " + currentBid + " FM!");

            } else {

                alert("L'offerta deve essere superiore a quella attuale!");

            }

        }


        // --- CHAT ---

        function sendChat() {

            let input = document.getElementById('chat-input');

            let user = document.getElementById('username-input').value;

            if(input.value.trim() !== "") {

                addChatMessage(user, input.value);

                input.value = "";

            }

        }


        function addChatMessage(sender, msg) {

            let container = document.getElementById('chat-messages');

            let div = document.createElement('div');

            div.className = 'chat-msg';

            div.innerHTML = `<b>${sender}:</b> ${msg}`;

            container.appendChild(div);

            container.scrollTop = container.scrollHeight;

        }

        const samplePlayers = [

            { name: "Lautaro Martinez", team: "Inter", role: "A", fvm: 90 },

            { name: "Nicolò Barella", team: "Inter", role: "C", fvm: 35 },

            { name: "Gleison Bremer", team: "Juventus", role: "D", fvm: 28 },

            { name: "Mike Maignan", team: "Milan", role: "P", fvm: 25 },

            { name: "Khvicha Kvaratskhelia", team: "Napoli", role: "A", fvm: 75 }

        ];


        function loadListoneData() {

            filterListone();

        }


        function filterListone() {

            const search = document.getElementById('search-listone').value.toLowerCase();

            const role = document.getElementById('filter-role').value;

            const sort = document.getElementById('sort-order').value;


            let filtered = samplePlayers.filter(p => {

                const matchName = p.name.toLowerCase().includes(search);

                const matchRole = role === "ALL" || p.role === role;

                return matchName && matchRole;

            });


            if(sort === "alpha") {

                filtered.sort((a,b) => a.name.localeCompare(b.name));

            } else {

                filtered.sort((a,b) => b.fvm - a.fvm);

            }


            const container = document.getElementById('listone-container');

            container.innerHTML = "";

            filtered.forEach(p => {

                container.innerHTML += `

                    <div class="list-item">

                        <div>

                            <b>${p.name}</b> <small>(${p.team})</small>

                            <br><span style="color:#00e676; font-size:0.8rem;">Ruolo: ${p.role}</span>

                        </div>

                        <div><b>${p.fvm} FM</b></div>

                    </div>

                `;

            });

        }
